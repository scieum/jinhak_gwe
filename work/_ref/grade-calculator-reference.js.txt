function clamp(value, min, max) {
    return Math.min(max, Math.max(min, value));
}

function numberOr(value, fallback) {
    var parsed = parseFloat(value);
    return Number.isFinite(parsed) ? parsed : fallback;
}

function format(value, digits) {
    return value.toFixed(digits);
}

function interpolate(value, inputs, outputs) {
    var v = clamp(value, inputs[0], inputs[inputs.length - 1]);
    if (v <= inputs[0]) return outputs[0];
    for (var i = 0; i < inputs.length - 1; i++) {
        if (v <= inputs[i + 1]) {
            var ratio = (v - inputs[i]) / (inputs[i + 1] - inputs[i]);
            return outputs[i] + (outputs[i + 1] - outputs[i]) * ratio;
        }
    }
    return outputs[outputs.length - 1];
}

function percentFromFive(value) {
    var model = models[state.model];
    if (state.model === "theory") return percentFromGradeAverage(value, fiveCenters, 5);
    return interpolate(clamp(value, 1, 5), model.five, model.percent);
}

function percentFromNine(value) {
    var model = models[state.model];
    if (state.model === "theory") return percentFromGradeAverage(value, nineCenters, 9);
    var grade = clamp(value, 1, 9);
    if (grade < model.nine[0]) {
        return interpolate(grade, [1, model.nine[0]], [model.nineFloorPercent, model.percent[0]]);
    }
    return interpolate(grade, model.nine, model.percent);
}

function fiveFromPercent(percent) {
    var model = models[state.model];
    if (state.model === "theory") return gradeFromPercent(percent, fiveCenters);
    return interpolate(percent, model.percent, model.five);
}

function nineFromPercent(percent) {
    var model = models[state.model];
    if (state.model === "theory") return gradeFromPercent(percent, nineCenters);
    if (percent < model.percent[0]) {
        return interpolate(percent, [model.nineFloorPercent, model.percent[0]], [1, model.nine[0]]);
    }
    return interpolate(percent, model.percent, model.nine);
}

function percentFromGradeAverage(value, centers, maxGrade) {
    var gradeValue = clamp(value, 1, maxGrade);
    var lower = Math.floor(gradeValue);
    var upper = Math.ceil(gradeValue);
    if (lower === upper) return centers[lower - 1];

    var lowerCenter = centers[lower - 1];
    var upperCenter = centers[upper - 1];
    var ratio = gradeValue - lower;
    return lowerCenter + (upperCenter - lowerCenter) * ratio;
}

function gradeFromPercent(percent, centers) {
    var p = clamp(percent, 0, 100);
    if (p <= centers[0]) return 1;
    if (p >= centers[centers.length - 1]) return centers.length;

    for (var i = 0; i < centers.length - 1; i++) {
        var left = centers[i];
        var right = centers[i + 1];
        if (p >= left && p <= right) {
            var ratio = (p - left) / (right - left);
            return (i + 1) + ratio;
        }
    }
    return centers.length;
}

function bandForPercent(bands, percent) {
    var p = clamp(percent, 0, 100);
    for (var i = 0; i < bands.length; i++) {
        if (p <= bands[i].end) return bands[i];
    }
    return bands[bands.length - 1];
}

function bandLabel(band) {
    return band.grade + "등급 · 상위 " + band.start + "~" + band.end + "%";
}

function overlapNineRangeForFiveBand(fiveBand) {
    var matched = nineBands.filter(function(nineBand) {
        return nineBand.end > fiveBand.start && nineBand.start < fiveBand.end;
    });
    if (matched.length === 0) return "대표값 기준";

    var first = matched[0].grade;
    var last = matched[matched.length - 1].grade;
    if (first === last) return "9등급제 " + first + "등급 범위";
    return "9등급제 " + first + "~" + last + "등급 범위";
}

function renderScale(container, bands, type) {
    container.innerHTML = "";
    bands.forEach(function(band) {
        var item = document.createElement("div");
        item.className = "scale-band g" + band.grade;
        item.style.flexBasis = band.share + "%";
        item.title = bandLabel(band);
        item.setAttribute("aria-label", bandLabel(band));
        item.innerHTML = "<span>" + (type === "nine" ? band.grade : band.grade + "등급") + "</span>";
        container.appendChild(item);
    });

    var marker = document.createElement("div");
    marker.className = "percent-marker";
    marker.setAttribute("aria-hidden", "true");
    container.appendChild(marker);
}

function setMarkers(percent) {
    var markers = document.querySelectorAll(".percent-marker");
    markers.forEach(function(marker) {
        marker.style.left = clamp(percent, 0, 100) + "%";
    });
}

// Only write a control if the user isn't actively editing it,
// so live recalculation never fights the field being typed in.
function writeControl(el, value) {
    if (el !== document.activeElement) el.value = value;
}

    }
    if (state.source === "percent") {
        return "상위 " + format(percent, 1) + "%는 5등급제 약 " + format(five, 2) + "등급, 9등급제 약 " + format(nine, 2) + "등급에 해당합니다.";
    }
    return "5등급제 " + format(five, 2) + "등급은 9등급제 약 " + format(nine, 2) + "등급(상위 " + format(percent, 1) + "%)에 해당합니다.";
}

    var nine = clamp(nineFromPercent(percent), 1, 9);
    var fiveBand = bandForPercent(fiveBands, percent);
    var nineBand = bandForPercent(nineBands, percent);

    writeControl(els.fiveInput, format(five, 2));
    writeControl(els.fiveSlider, format(five, 2));
    writeControl(els.nineInput, format(nine, 2));
    writeControl(els.nineSlider, format(nine, 2));
    writeControl(els.percentInput, format(percent, 1));
    writeControl(els.percentSlider, format(percent, 1));

    els.resultFive.textContent = format(five, 2);
    els.resultNine.textContent = format(nine, 2);
    els.resultSummary.textContent = summaryText(five, nine, percent);
    els.percentValue.textContent = "상위 " + format(percent, 1) + "%";
    els.fiveRange.textContent = bandLabel(fiveBand);
    els.nineRange.textContent = bandLabel(nineBand);
    els.rangeHint.textContent = overlapNineRangeForFiveBand(fiveBand);
    els.fiveBandText.textContent = fiveBand.grade + "등급 구간";
    els.nineBandText.textContent = nineBand.grade + "등급 구간";
    var model = models[state.model];
    els.resultKicker.textContent = model.label + " · 변환 결과";
    els.modelDescription.textContent = model.description;
    els.methodText.textContent = model.method;
    els.sourceLink.href = model.source;
    setMarkers(percent);

    els.rows.forEach(function(row) {
        row.classList.toggle("is-source", row.getAttribute("data-axis") === state.source);
    });
}

}

function fromPercent(value) {
